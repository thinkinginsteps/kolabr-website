    const base = 'rotateY(-19deg) rotateX(5deg) rotate(.8deg)';
    const ease = 'cubic-bezier(.16,.84,.28,1)';
    const steps = [
      // arriving in front: leads the move, comes up to full opacity fast
      { t: `translateY(-50%) ${base} translate3d(0,0,0) scale(1)`, o: 1, z: 3,
        tr: `transform .95s ${ease}, opacity .4s ease` },
      // waiting on deck
      { t: `translateY(-44%) ${base} translate3d(3%,0,-110px) scale(.975)`, o: .6, z: 2,
        tr: `transform .95s ${ease} .12s, opacity .5s ease .25s` },
      // just left the front: fades out where it stands, then slips back unseen
      { t: `translateY(-38%) ${base} translate3d(6%,0,-220px) scale(.95)`, o: .32, z: 1,
        tr: `transform .8s ${ease} .34s, opacity .3s ease` }
    ];
    const out = {};
    for (let i = 0; i < 3; i++) {
      const pos = (i - this.state.shot + 3) % 3, s = steps[pos];
      out['tf' + i] = s.t; out['op' + i] = s.o; out['z' + i] = s.z; out['pos' + i] = String(pos); out['tr' + i] = s.tr;
    }
    return out;
  }

      compareBg: m === 'compare' ? on : 'transparent',

    els.forEach(el => io.observe(el));
